CXX = g++
CXXFLAGS = -std=c++17 -pthread -Iincludes
TARGET = lab3
SRC_DIR = src
OBJ_DIR = obj
BIN_DIR = .

SOURCES = $(SRC_DIR)/main.cpp $(SRC_DIR)/marker_array.cpp $(SRC_DIR)/marker_controller.cpp
OBJECTS = $(OBJ_DIR)/main.o $(OBJ_DIR)/marker_array.o $(OBJ_DIR)/marker_controller.o

all: directories $(BIN_DIR)/$(TARGET)

directories:
	mkdir -p $(OBJ_DIR)

$(BIN_DIR)/$(TARGET): $(OBJECTS)
	$(CXX) $(CXXFLAGS) -o $@ $^

$(OBJ_DIR)/main.o: $(SRC_DIR)/main.cpp
	$(CXX) $(CXXFLAGS) -c $< -o $@

$(OBJ_DIR)/marker_array.o: $(SRC_DIR)/marker_array.cpp includes/marker_array.h
	$(CXX) $(CXXFLAGS) -c $< -o $@

$(OBJ_DIR)/marker_controller.o: $(SRC_DIR)/marker_controller.cpp includes/marker_controller.h includes/marker_array.h
	$(CXX) $(CXXFLAGS) -c $< -o $@

run: $(BIN_DIR)/$(TARGET)
	./$(BIN_DIR)/$(TARGET)

clean:
	rm -rf $(OBJ_DIR) $(BIN_DIR)/$(TARGET)

.PHONY: all directories run clean
